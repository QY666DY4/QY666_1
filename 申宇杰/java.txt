import com.alibaba.fastjson2.JSON;
import com.alibaba.fastjson2.JSONObject;
import okhttp3.*;
import java.io.IOException;

public class LLMAPICaller {
    private static final String API_URL = "https://open.bigmodel.cn/api/paas/v4/chat/completions";
    private static final String API_KEY = System.getenv("ZAI_API_KEY");

    public static void main(String[] args) {
        OkHttpClient client = new OkHttpClient();
        String jsonBody = "{\"model\":\"glm-4\",\"messages\":[{\"role\":\"user\",\"content\":\"你好，请介绍一下你自己\"}]}";

        Request request = new Request.Builder()
            .url(API_URL)
            .post(RequestBody.create(jsonBody, MediaType.get("application/json; charset=utf-8")))
            .addHeader("Authorization", "Bearer " + API_KEY)
            .build();

        try (Response response = client.newCall(request).execute()) {
            JSONObject res = JSON.parseObject(response.body().string());
            System.out.println("AI: " + res.getJSONArray("choices").getJSONObject(0).getJSONObject("message").getString("content"));
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
client.newCall(request).enqueue(object : Callback {
    override fun onResponse(call: Call, response: Response) {
        response.body?.source()?.use { source ->
            while (!source.exhausted()) {
                val line = source.readUtf8Line()
                if (line?.startsWith("data: ") == true && line != "data: [DONE]") {
                    print(parseContent(line.substring(6)))
                }
            }
        }
    }
})

// 2. 构建请求
ChatCompletionCreateParams req = ChatCompletionCreateParams
    .builder().model("glm-4")
    .messages(Arrays.asList(
        ChatMessage.builder().role("user")
            .content("你好，请介绍一下你自己").build()
    )).build();

// 3. 发送请求
client.chat().createChatCompletion(req)
    .filter(Response::isSuccess)
    .ifPresent(res -> {
        String reply = res.getData().getChoices()

        System.out.println("AI 回复: " + reply);
